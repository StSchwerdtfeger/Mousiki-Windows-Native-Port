##             STEREO
##-------------------------------------------

StereoPlayback=true
## true  = play tracks in stereo (uses about twice the memory per loaded track)
## false = mono: left and right are folded together, as in earlier versions
## Turning it off takes effect immediately; turning it on applies from the next track.

##-------------------------------------------
##             LOUDNESS NORMALIZATION
##-------------------------------------------

NormalizeVolume=true
NormalizeTargetLufs=-16
NormalizeMaxBoostDb=9
## Each track is measured (LUFS) while it decodes and played at NormalizeTargetLufs.
## Quiet tracks are raised (at most NormalizeMaxBoostDb), loud/compressed ones lowered.
## -14 matches YouTube/Spotify; -16 leaves more headroom. Lower number = quieter overall.

##-------------------------------------------

ConsoleVerbosity=basic

## basic   = every command mousiki ran (yt-dlp/ffprobe/ffmpeg/lyrics-fetch) + its raw output
## verbose = basic, plus internal/OS-level events (terminal resize, audio device init, spawn errors, ...)
## Log file (Windows): %USERPROFILE%\.cache\mousiki\logs\console.log -- wiped fresh at the start of every session.
